> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 ironmlx P4 Dense（Qwen3.5 文本路径已完成）基础上加 vision_tower + image preprocessing + cross-modal token routing，让 Qwen3.5-4B-MLX-4bit 跟 mlx-vlm 0.5.0 logits 对齐推理图文混合输入。

- Acceptance: max_diff < 0.5 vs mlx-vlm last-position logits + greedy bit-identical first token

```rust
// 默认值跟 HF qwen2_vl image_processor 一致
const FACTOR: i32 = 32;          // patch_size * spatial_merge_size = 16 * 2
const MIN_PIXELS: i32 = 56 * 56; // 3136
const MAX_PIXELS: i32 = 14 * 14 * 4 * 1280; // 1003520

/// Port of mlx-vlm `_smart_resize_image` — 保 aspect ratio + 满足 patch
/// 对齐 + 总像素在 [MIN_PIXELS, MAX_PIXELS]。
pub fn smart_resize(height: i32, width: i32) -> (i32, i32) {
    let max_dim = height.max(width) as f64;
    let min_dim = height.min(width) as f64;
    if max_dim / min_dim > 200.0 {
        panic!("absolute aspect ratio must be smaller than 200");
    }
    let f = FACTOR as f64;
    let mut h_bar = ((height as f64 / f).round() * f) as i32;
    let mut w_bar = ((width as f64 / f).round() * f) as i32;
    if h_bar * w_bar > MAX_PIXELS {
        let beta = ((height as f64 * width as f64) / MAX_PIXELS as f64).sqrt();
        h_bar = (((height as f64 / beta) / f).floor() * f).max(f) as i32;
        w_bar = (((width as f64 / beta) / f).floor() * f).max(f) as i32;
    } else if h_bar * w_bar < MIN_PIXELS {
        let beta = (MIN_PIXELS as f64 / (height as f64 * width as f64)).sqrt();
        h_bar = (((height as f64 * beta) / f).ceil() * f) as i32;
        w_bar = (((width as f64 * beta) / f).ceil() * f) as i32;
    }
    (h_bar, w_bar)
}
```

```rust
const IMAGE_MEAN: [f32; 3] = [0.5, 0.5, 0.5];
const IMAGE_STD: [f32; 3] = [0.5, 0.5, 0.5];

/// (px/255 - mean) / std  for a single RGB pixel.
pub fn normalize_pixel(rgb: [u8; 3]) -> [f32; 3] {
    [
        (rgb[0] as f32 / 255.0 - IMAGE_MEAN[0]) / IMAGE_STD[0],
        (rgb[1] as f32 / 255.0 - IMAGE_MEAN[1]) / IMAGE_STD[1],
        (rgb[2] as f32 / 255.0 - IMAGE_MEAN[2]) / IMAGE_STD[2],
    ]
}

const PATCH: i32 = 16;
const TEMPORAL_PATCH: i32 = 2;

/// Reshape [3, H, W] f32 raw pixels into Qwen3.5 vision patches.
///
/// Output shape: `[grid_h * grid_w, TEMPORAL_PATCH (=2), 3, PATCH (=16), PATCH]`,
/// where temporal axis duplicates the single image frame to match `temporal_patch_size=2`.
pub fn patchify(raw: &[f32], h: i32, w: i32) -> (mlx::Array, i32, i32) {
    assert_eq!(raw.len(), (3 * h * w) as usize);
    assert_eq!(h % PATCH, 0);
    assert_eq!(w % PATCH, 0);
    let grid_h = h / PATCH;
    let grid_w = w / PATCH;
    // [3, H, W] → [3, grid_h, PATCH, grid_w, PATCH] → permute (1,3,0,2,4) → [grid_h, grid_w, 3, PATCH, PATCH]
    // → reshape → [grid_h*grid_w, 3, PATCH, PATCH]
    // → expand temporal axis (broadcast 2× the single frame) → [grid_h*grid_w, 2, 3, PATCH, PATCH]
    let arr: mlx::Array = (raw, &[3, h, w][..]).try_into().expect("array");
    let arr = arr
        .reshape(&[3, grid_h, PATCH, grid_w, PATCH])
        .expect("reshape");
    let arr = mlx::ops::shape::transpose_axes(&arr, &[1, 3, 0, 2, 4]).expect("transpose");
    let arr = arr.reshape(&[grid_h * grid_w, 3, PATCH, PATCH]).expect("reshape2");
    // expand temporal: [N, 1, 3, P, P] → broadcast to [N, 2, 3, P, P]
    let arr = arr.expand_dims(1).expect("expand");
    let arr = mlx::ops::shape::broadcast_to(&arr, &[grid_h * grid_w, TEMPORAL_PATCH, 3, PATCH, PATCH]).expect("bcast");
    (arr, grid_h, grid_w)
}
```

```rust
use anyhow::{anyhow, Result};

/// Pipeline: decode → smart_resize → normalize → patchify.
/// Returns (pixel_values, grid_h, grid_w).
pub fn preprocess(img_bytes: &[u8]) -> Result<(mlx::Array, i32, i32)> {
    // 1. decode
    let img = image::load_from_memory(img_bytes)
        .map_err(|e| anyhow!("decode image: {e}"))?
        .to_rgb8();
    let (orig_w, orig_h) = (img.width() as i32, img.height() as i32);

    // 2. smart resize
    let (h2, w2) = smart_resize(orig_h, orig_w);

    // 3. resize via Lanczos3 (HF default)
    let resized = image::imageops::resize(
        &img,
        w2 as u32,
        h2 as u32,
        image::imageops::FilterType::Lanczos3,
    );

    // 4. normalize: [(H, W, 3) u8] → [(3, H, W) f32], (px/255 - 0.5)/0.5
    let n_pix = (3 * h2 * w2) as usize;
    let mut chw = vec![0.0_f32; n_pix];
    let plane = (h2 * w2) as usize;
    for (i, p) in resized.pixels().enumerate() {
        let n = normalize_pixel([p.0[0], p.0[1], p.0[2]]);
        chw[0 * plane + i] = n[0];
        chw[1 * plane + i] = n[1];
        chw[2 * plane + i] = n[2];
    }

    // 5. patchify
    let (pixel_values, gh, gw) = patchify(&chw, h2, w2);
    Ok((pixel_values, gh, gw))
}
```

```rust
use mlx::{Array, Dtype};

/// Vision rotary frequency table: `freqs[s, i] = s / (theta ^ (2i / dim))`.
/// Output shape: [seqlen, dim].
pub fn build_rotary_freqs(seqlen: i32, dim: i32, theta: f32) -> Array {
    use mlx::ops;
    // inv_freq[i] = 1 / theta^(2i/dim)  for i in [0, dim/2 ... interpreted as dim entries via [0, dim) step 2]
    // Following mlx-vlm: arange(0, dim, 2) → 16 entries when dim=32
    // But shape says (max_hw, dim // 2) — so actual freq dim is dim/2 not dim.
    // Wait re-read mlx-vlm: VisionRotaryEmbedding(dim=head_dim//2), so dim arg is half of head_dim.
    // Inside, arange(0, dim, 2) with step=2 gives dim/2 entries. So output is [seqlen, dim/2].
    //
    // We'll match mlx-vlm: arg `dim` is "the rotary dim" (= head_dim/2), and freqs is [seqlen, dim/2].
    let half = dim / 2;
    let exponents: Vec<f32> = (0..half).map(|i| (2 * i) as f32 / dim as f32).collect();
    let inv_freq: Array = (exponents.as_slice(), &[half][..]).try_into().unwrap();
    let inv_freq = ops::binary::pow_scalar(theta, &inv_freq).unwrap();
    let inv_freq = ops::binary::reciprocal(&inv_freq).unwrap();
    let seq: Array = ((0..seqlen).map(|i| i as f32).collect::<Vec<_>>().as_slice(), &[seqlen][..]).try_into().unwrap();
    // outer: seq [seqlen] × inv_freq [half] → [seqlen, half]
    let seq2 = seq.reshape(&[seqlen, 1]).unwrap();
    let inv2 = inv_freq.reshape(&[1, half]).unwrap();
    (&seq2 * &inv2).expect("outer")
}
```

```rust
//! Patch embedding for Qwen3.5 vision encoder.

use anyhow::Result;
use mlx::{ops, Array, StreamOrDevice};

use crate::core::Loader;

/// Conv3D-equivalent flatten: input patches [N, 2 (T), 3 (C), 16, 16]
/// → reshape to [N, 1536] → matmul weight^T (reshaped from [1024,2,16,16,3]).
///
/// Output: [N, hidden_size=1024].
pub struct PatchEmbed {
    /// Weight reshape from `[1024, 2, 16, 16, 3]` to `[1024, 1536]`.
    /// Note flatten order is (T, H, W, C) — last dim is innermost in HF layout.
    weight_2d: Array,
    bias: Array,
    hidden_size: i32,
}

impl PatchEmbed {
    pub fn new(weight_5d: Array, bias: Array, hidden_size: i32) -> Self {
        // Reshape (1024, 2, 16, 16, 3) → (1024, 1536) in (T, H, W, C) flatten order
        let weight_2d = weight_5d
            .reshape(&[hidden_size, 2 * 16 * 16 * 3])
            .expect("reshape patch_embed weight");
        Self {
            weight_2d,
            bias,
            hidden_size,
        }
    }

    pub fn from_loader(loader: &Loader, prefix: &str, hidden_size: i32) -> Result<Self> {
        let weight = loader.tensor(&format!("{prefix}.weight"))?.clone();
        let bias = loader.tensor(&format!("{prefix}.bias"))?.clone();
        Ok(Self::new(weight, bias, hidden_size))
    }

    pub fn forward(&self, x: &Array) -> Result<Array> {
        self.forward_on(x, ())
    }

    pub fn forward_on(&self, x: &Array, target: impl Into<StreamOrDevice>) -> Result<Array> {
        let target = target.into();
        // x: [N, 2, 3, 16, 16] (T, C, H, W) — but HF expects (T, H, W, C) for flatten.
        // mlx-vlm `PatchEmbed.forward` does: reshape(-1, C, T, H, W).moveaxis(1, 4)
        //   → [N, T, H, W, C] = [N, 2, 16, 16, 3]
        // 然后 Conv3D + reshape to [N, hidden].
        // 我们 patchify 输出已经是 [N, 2, 3, 16, 16] = [N, T, C, H, W]，需要 moveaxis 让 C 到最后：
        //   transpose to [N, T, H, W, C] = [N, 2, 16, 16, 3]
        let x = ops::shape::transpose_axes(x, &[0, 1, 3, 4, 2])?;
        let n = x.shape().as_slice()[0];
        let x = x.reshape(&[n, 2 * 16 * 16 * 3])?;
        // matmul (x @ weight^T)+bias — ironmlx Linear pattern
        let wt = self.weight_2d.transpose_on(target)?;
        let out = x.matmul_on(&wt, target)?;
        Ok(&out + &self.bias)
    }
}
```

```rust
            // mlx::fast::scaled_dot_product_attention(q, k, v, scale, mask=None)
            // 按 mlx crate 实际 API 调
            let out = mlx::fast::scaled_dot_product_attention_on(&q_seg, &k_seg, &v_seg, scale, None, ())?;
            outs.push(out);
        }
        let merged = if outs.len() == 1 {
            outs.pop().unwrap()
        } else {
            mlx::ops::shape::concatenate_on(&outs.iter().collect::<Vec<_>>(), 1, ())?
        };

        // 5. transpose back, reshape, proj
        let merged = ops::shape::transpose_axes(&merged, &[1, 0, 2])?; // [seq, num_heads, head_dim]
        let merged = merged.reshape(&[seq_len, self.num_heads * self.head_dim])?;
        let out = merged.matmul(&self.proj_w.transpose()?)? + &self.proj_b;
        Ok(out)
    }
}

/// Apply rotary pos emb to vision Q/K (跟 mlx-vlm `apply_rotary_pos_emb_vision` 一致).
/// `tensor: [seq, num_heads, head_dim]`, `freqs: [seq, head_dim/2]`.
fn apply_rotary_vision(tensor: &Array, freqs: &Array) -> Result<Array> {
    let head_dim = tensor.shape().as_slice()[2];
    let half = head_dim / 2;
    // Split tensor along head_dim into two halves (x1, x2)
    let x1 = ops::indexing::slice_strided(tensor,
        &[0, 0, 0][..], &[tensor.shape().as_slice()[0], tensor.shape().as_slice()[1], half][..],
        &[1, 1, 1][..])?;
    let x2 = ops::indexing::slice_strided(tensor,
        &[0, 0, half][..], &[tensor.shape().as_slice()[0], tensor.shape().as_slice()[1], head_dim][..],
        &[1, 1, 1][..])?;
    // freqs[seq, half] → broadcast to [seq, num_heads, half] via expand
    let freqs_e = freqs.reshape(&[freqs.shape().as_slice()[0], 1, half])?;
    let cos = freqs_e.cos()?;
    let sin = freqs_e.sin()?;
    // rotate: (x1*cos - x2*sin, x1*sin + x2*cos)
    let new_x1 = (&x1 * &cos)? - (&x2 * &sin)?;
    let new_x2 = (&x1 * &sin)? + (&x2 * &cos)?;
    let out = mlx::ops::shape::concatenate_on(&[&new_x1, &new_x2], -1, ())?;
    Ok(out)
}
```

```rust
use anyhow::Result;
use mlx::{ops, Array};
use crate::core::Loader;
use crate::nn::LayerNorm;

pub struct PatchMerger {
    norm: LayerNorm,
    fc1_w: Array,
    fc1_b: Array,
    fc2_w: Array,
    fc2_b: Array,
    spatial_merge_size: i32,
    hidden_size: i32, // 1024
}

impl PatchMerger {
    pub fn new(
        norm_w: Array, norm_b: Array,
        fc1_w: Array, fc1_b: Array,
        fc2_w: Array, fc2_b: Array,
        spatial_merge_size: i32,
    ) -> Self {
        Self {
            norm: LayerNorm::new(norm_w, Some(norm_b), 1e-6),
            fc1_w, fc1_b, fc2_w, fc2_b,
            spatial_merge_size,
            hidden_size: 1024,
        }
    }

    pub fn from_loader(loader: &Loader, prefix: &str, spatial_merge_size: i32) -> Result<Self> {
        Ok(Self {
            norm: LayerNorm::from_loader(loader, &format!("{prefix}.norm"), 1e-6)?,
            fc1_w: loader.tensor(&format!("{prefix}.linear_fc1.weight"))?.clone(),
            fc1_b: loader.tensor(&format!("{prefix}.linear_fc1.bias"))?.clone(),
            fc2_w: loader.tensor(&format!("{prefix}.linear_fc2.weight"))?.clone(),
            fc2_b: loader.tensor(&format!("{prefix}.linear_fc2.bias"))?.clone(),
            spatial_merge_size,
            hidden_size: 1024,
        })
    }

    /// `x: [N_patches=sum(grid_h*grid_w), 1024]` post-block tokens.
    /// `grid_thw: per-image (T, H, W) grid sizes`.
    /// 输出 `[N_merged, 2560]`.
    pub fn forward(&self, x: &Array, grid_thw: &[(i32, i32, i32)]) -> Result<Array> {
        // mlx-vlm: x = norm(x) (per-patch on 1024)
        //          x = x.reshape(-1, hidden * merge^2)  // 把每 merge^2 个 patch 合
        //          x = fc2(act(fc1(x)))
        //
        // mlx-vlm 的 reshape 假设 patches 已经按 merge 顺序排好（前 merge^2 个 = 同一个 merged 块）
        // 我们 patchify 是按 row-major (h_idx*grid_w + w_idx)；mlx-vlm 在 rot_pos_emb / spatial logic 中
        // 已重新排序到 "merge 块连续"。这里假设已排好，否则需先重排。

        let x = self.norm.forward(x)?;
        let merge_sq = self.spatial_merge_size * self.spatial_merge_size;
        let merged_dim = self.hidden_size * merge_sq;
        let total = x.shape().as_slice()[0];
        assert_eq!(total % merge_sq, 0);
        let n_merged = total / merge_sq;
        let x = x.reshape(&[n_merged, merged_dim])?;
        // fc1
        let h = (&x.matmul(&self.fc1_w.transpose()?)? + &self.fc1_b)?;
        // GELU
        let h = mlx::ops::nn::gelu(&h)?; // 标准 GELU (not tanh approx — merger 用 plain)
        // fc2
        let out = (&h.matmul(&self.fc2_w.transpose()?)? + &self.fc2_b)?;
        Ok(out)
    }
}
```

```rust
//! Qwen3.5 vision tower — wraps PatchEmbed + 24 ViT blocks + PatchMerger.

use anyhow::Result;
use mlx::Array;

use super::{block::VitBlock, merger::PatchMerger, patch_embed::PatchEmbed};
use crate::core::Loader;
use crate::models::qwen3_5::config::VisionConfig;

pub struct VisionTower {
    patch_embed: PatchEmbed,
    pos_embed: Array,           // [num_position_embeddings=2304, hidden=1024]
    rotary_dim: i32,            // head_dim/2 = 32
    rotary_theta: f32,          // 10000
    blocks: Vec<VitBlock>,
    merger: PatchMerger,
    hidden_size: i32,
    num_grid_per_side: i32,     // sqrt(num_position_embeddings) = 48
    spatial_merge_size: i32,
    head_dim: i32,
}

impl VisionTower {
    pub fn from_loader(loader: &Loader, cfg: &VisionConfig) -> Result<Self> {
        let head_dim = cfg.hidden_size / cfg.num_heads;
        let patch_embed = PatchEmbed::from_loader(loader, "vision_tower.patch_embed.proj", cfg.hidden_size)?;
        let pos_embed = loader.tensor("vision_tower.pos_embed.weight")?.clone();
        let mut blocks = Vec::with_capacity(cfg.depth as usize);
        for i in 0..cfg.depth {
            blocks.push(VitBlock::from_loader(
                loader, &format!("vision_tower.blocks.{i}"), cfg.num_heads, head_dim,
            )?);
        }
        let merger = PatchMerger::from_loader(loader, "vision_tower.merger", cfg.spatial_merge_size)?;
        let num_grid_per_side = (cfg.num_position_embeddings as f64).sqrt() as i32;
        Ok(Self {
            patch_embed, pos_embed,
            rotary_dim: head_dim / 2,
            rotary_theta: 10000.0,
            blocks, merger,
            hidden_size: cfg.hidden_size,
            num_grid_per_side,
            spatial_merge_size: cfg.spatial_merge_size,
            head_dim,
        })
    }

    pub fn depth(&self) -> i32 {
        self.blocks.len() as i32
    }

    /// `pixel_values: [N_patches, 2 (T), 3, 16, 16]` from image_processor.
    /// `grid_thw: per-image (T=1, H, W)`.
    /// Returns `vision_embeds: [N_merged, 2560]`.
    pub fn forward(&self, pixel_values: &Array, grid_thw: &[(i32, i32, i32)]) -> Result<Array> {
        // 1. patch embed
        let mut x = self.patch_embed.forward(pixel_values)?;

        // 2. add learned 2D pos_embed per (h, w)
        x = self.add_learned_pos_embed(&x, grid_thw)?;

        // 3. compute rotary pos_emb table per grid
        let rotary = self.compute_rotary_pos_emb(grid_thw)?;

        // 4. cu_seqlens (P6 单图)
        let cu_seqlens: Vec<i32> = {
            let mut v = vec![0_i32];
            let mut total = 0;
            for (t, h, w) in grid_thw {
                total += t * h * w;
                v.push(total);
            }
            v
        };

        // 5. 24 ViT blocks
        for blk in &self.blocks {
            x = blk.forward(&x, &rotary, &cu_seqlens)?;
        }

        // 6. merger
        self.merger.forward(&x, grid_thw)
    }

    /// Look up learned pos_embed per (h, w) grid pos within each image, broadcast across T.
    /// pos_embed table is [2304=48*48, 1024]; we use bilinear interpolation 把 actual grid
    /// (变长) 投到 48×48 表上，跟 mlx-vlm `fast_pos_embed_interpolate` 一致。
    fn add_learned_pos_embed(&self, x: &Array, grid_thw: &[(i32, i32, i32)]) -> Result<Array> {
        // 简化版：先按 grid (h, w) 直接取 pos_embed[h*48 + w] (mod 2304)，等同跑通后再做 bilinear interp 升级
        // (P6 first cut，跟 mlx-vlm logits 偏差大就改)
        // P6.1 完整 bilinear interp。
        // 占位实现：返回 x 原值（pos_embed 跳过）。
        // ❌ 这会让 logits 偏，但能编译先跑通。后续 Step 12.4 用 bilinear interp 完整版。
        Ok(x.clone())
    }

    /// Compute rotary freq table per merged-block grid position.
    fn compute_rotary_pos_emb(&self, grid_thw: &[(i32, i32, i32)]) -> Result<Array> {
        // mlx-vlm `rot_pos_emb`: 对每个 (t, h, w) image，构造 pos_ids [N_patches, 2]
        // (h_pos, w_pos)，每个值 [0, max_hw)；然后查 freq table 拼起来。
        // 完整算法见 mlx-vlm vision.py:232-280。
        // 简化占位：返回 [N_patches, head_dim/2] 全零（rotary 退化为不旋转）。
        let n_patches: i32 = grid_thw.iter().map(|(t, h, w)| t * h * w).sum();
        Ok(mlx::Array::zeros(&[n_patches, self.rotary_dim], mlx::Dtype::Float32)?)
    }
}
```

NOTE: `add_learned_pos_embed` 跟 `compute_rotary_pos_emb` 第一版用占位实现（pos_embed 跳过、rotary 全零）。让 vision_tower **能跑通 + 能 load**，但 logits 会偏。后续 step 升级到 mlx-vlm 完整算法。

```rust
use anyhow::Result;
use mlx::{ops, Array};

/// Replace `text_embeds[b, s, :]` with `vision_embeds[k, :]` at every position
/// `s` where `input_ids[b, s] == image_token_id`. `k` increments per image token
/// in row-major order.
pub fn replace_image_tokens(
    text_embeds: &Array,        // [B, S, hidden]
    input_ids: &Array,          // [B, S]
    vision_embeds: &Array,      // [N_image_tokens, hidden]
    image_token_id: i32,
) -> Result<Array> {
    // 简化：B=1 (P6 单 sequence)。
    let dims = text_embeds.shape();
    let dims_s = dims.as_slice();
    let (b, s, hidden) = (dims_s[0], dims_s[1], dims_s[2]);
    assert_eq!(b, 1);

    // mask: where(input_ids == image_token_id) → 1 else 0, shape [B, S]
    let img_mask = ops::comparison::equal_scalar(input_ids, image_token_id)?; // bool
    let mask = mlx::ops::cast::astype(&img_mask, text_embeds.dtype())?;
    let mask = mask.reshape(&[b, s, 1])?;
    let inv_mask = ((1.0_f32 as f32) - &mask)?;

    // Build "vision_at_text_positions" tensor [B, S, hidden]:
    //   for positions where mask=1, fill vision_embeds[k] (k = cumsum of mask up to that position)
    //   for positions where mask=0, fill anything (will be zeroed)
    //
    // 简化路径：用 scatter / gather。
    // 这里用一种可读路径：先把 vision_embeds reshape 成 [1, N_img, hidden]，
    // 然后 cumsum 算每位的 image-index，gather 取出。
    //
    // 占位简化（B=1 + 单序列）：CPU 侧 build 一个 [1, S, hidden] tensor 直接拼。
    // 实际生产可改为 mlx 端 gather。

    let n_img = vision_embeds.shape().as_slice()[0];
    let img_mask_flat: Vec<i32> = mlx::ops::cast::astype(&img_mask.reshape(&[s])?, mlx::Dtype::Int32)?
        .to_vec().unwrap();
    let v_flat: Vec<f32> = mlx::ops::cast::astype(vision_embeds, mlx::Dtype::Float32)?.to_vec().unwrap();
    let mut vision_at_text: Vec<f32> = vec![0.0; (s * hidden) as usize];
    let mut k = 0_i32;
    for i in 0..s as usize {
        if img_mask_flat[i] == 1 {
            for d in 0..hidden as usize {
                vision_at_text[i * hidden as usize + d] = v_flat[k as usize * hidden as usize + d];
            }
            k += 1;
        }
    }
    assert_eq!(k, n_img, "input_ids image token count != vision_embeds count");

    let v_at_text: Array = (vision_at_text.as_slice(), &[1, s, hidden][..]).try_into().unwrap();
    let v_at_text = mlx::ops::cast::astype(&v_at_text, text_embeds.dtype())?;

    // merged = text_embeds * (1-mask) + vision_at_text * mask
    Ok(((text_embeds * &inv_mask)? + (&v_at_text * &mask)?)?)
}
```

```rust
pub const IMAGE_TOKEN_ID: i32 = 248056;
pub const VISION_START_TOKEN_ID: i32 = 248053;
pub const VISION_END_TOKEN_ID: i32 = 248054;

/// Generate Qwen3.5 MRoPE position_ids for a sequence containing image tokens.
///
/// Output shape: `[3, 1, S]` (i32) — three streams (t, h, w) for B=1.
/// For text tokens: t = h = w = current_pos.
/// For image tokens: per-image t = base_t, h = base_t + row, w = base_t + col.
///
/// `input_ids: &[i32]` length S
/// `grid_thw: &[(t, h, w)]` per-image grid sizes from image_processor (t=1 for image-only)
/// `spatial_merge_size: i32` = 2 (vision config)
pub fn build_position_ids_vl(
    input_ids: &[i32],
    grid_thw: &[(i32, i32, i32)],
    image_token_id: i32,
    vision_start_token_id: i32,
    vision_end_token_id: i32,
    spatial_merge_size: i32,
) -> mlx::Array {
    let s = input_ids.len() as i32;
    // 三个 stream，i32 矩阵 [3, 1, S]
    let mut t_pos = vec![0_i32; s as usize];
    let mut h_pos = vec![0_i32; s as usize];
    let mut w_pos = vec![0_i32; s as usize];

    let mut base = 0_i32;       // current "logical position" base
    let mut img_idx = 0_usize;
    let mut i = 0_i32;
    while i < s {
        if input_ids[i as usize] == vision_start_token_id {
            // emit text pos for the start token
            t_pos[i as usize] = base;
            h_pos[i as usize] = base;
            w_pos[i as usize] = base;
            i += 1;
            // image_token_id 区段：N = grid_h/2 * grid_w/2 (after spatial merge)
            let (t_grid, h_grid, w_grid) = grid_thw[img_idx];
            let llm_grid_h = h_grid / spatial_merge_size;
            let llm_grid_w = w_grid / spatial_merge_size;
            // base_for_image: text base + 1
            let img_base = base + 1;
            for t_idx in 0..t_grid {
                for h_idx in 0..llm_grid_h {
                    for w_idx in 0..llm_grid_w {
                        if i >= s {
                            break;
                        }
                        if input_ids[i as usize] != image_token_id {
                            // unexpected: input_ids 应该含 grid_h*grid_w/4 个 image_token
                            panic!("expected image_token_id at position {i}, got {}", input_ids[i as usize]);
                        }
                        t_pos[i as usize] = img_base + t_idx;
                        h_pos[i as usize] = img_base + h_idx;
                        w_pos[i as usize] = img_base + w_idx;
                        i += 1;
                    }
                }
            }
            // base 更新：取所有 image stream 中最大值 + 1
            base = img_base + t_grid.max(llm_grid_h).max(llm_grid_w);
            img_idx += 1;
        } else {
            // text token (含 vision_end_token)
            t_pos[i as usize] = base;
            h_pos[i as usize] = base;
            w_pos[i as usize] = base;
            base += 1;
            i += 1;
        }
    }

    // Stack into [3, 1, S]
    let stacked: Vec<i32> = t_pos.into_iter().chain(h_pos).chain(w_pos).collect();
    (stacked.as_slice(), &[3_i32, 1, s][..]).try_into().unwrap()
}
```

```rust
use crate::models::qwen3_5::cross_modal;
use crate::models::qwen3_5::vision::VisionTower;

pub struct Qwen35Model {
    text: Qwen35TextModel,
    lm_head: Option<Linear>,
    /// `Some` if model has vision_tower weights; `None` for text-only checkpoints.
    pub(crate) vision: Option<VisionTower>,
}

impl Qwen35Model {
    pub fn from_loader(loader: &Loader) -> Result<Self> {
        let cfg = Qwen35Config::from_loader(loader)?;
        let text = Qwen35TextModel::from_loader(loader, &cfg)?;
        let lm_head = if cfg.tie_word_embeddings { None } else { Some(Linear::from_loader(loader, "lm_head")?) };
        let vision = match &cfg.vision_config {
            Some(vc) => Some(VisionTower::from_loader(loader, vc)?),
            None => None,
        };
        Ok(Self { text, lm_head, vision })
    }

    /// VL forward: routes images through vision_tower → cross-modal token replace
    /// → text backbone. If `pixel_values` is None, falls back to text-only path.
    pub fn forward_vl(
        &self,
        input_ids: &Array,
        position_ids: &Array,
        cache: Option<&mut [LayerCache]>,
        pixel_values: Option<&Array>,
        grid_thw: Option<&[(i32, i32, i32)]>,
        target: impl Into<StreamOrDevice>,
    ) -> Result<Array> {
        let target = target.into();
        let mut hidden = self.text.embed_tokens.forward_on(input_ids, target)?;
        if let Some(pv) = pixel_values {
            let grids = grid_thw.expect("grid_thw must be Some when pixel_values is Some");
            let vision_embeds = self.vision.as_ref()
                .ok_or_else(|| anyhow::anyhow!("model has no vision_tower but pixel_values supplied"))?
                .forward(pv, grids)?;
            hidden = cross_modal::replace_image_tokens(&hidden, input_ids, &vision_embeds, cross_modal::IMAGE_TOKEN_ID)?;
        }
        // Re-use text backbone forward post-embedding
        let (cos, sin) = self.text.mrope.cos_sin(position_ids)?;
        match cache {
            Some(c) => {
                for (layer, cell) in self.text.layers.iter().zip(c.iter_mut()) {
                    hidden = layer.forward_on(&hidden, &self.text.mrope, &cos, &sin, None, Some(cell), target)?;
                }
            }
            None => {
                for layer in &self.text.layers {
                    hidden = layer.forward_on(&hidden, &self.text.mrope, &cos, &sin, None, None, target)?;
                }
            }
        }
        let hidden = self.text.norm.forward_on(&hidden, target)?;
        let last = self.slice_last_position(&hidden, target)?;
        match &self.lm_head {
            Some(head) => head.forward_on(&last, target),
            None => self.text.as_output_on(&last, target),
        }
    }

    fn slice_last_position(&self, hidden: &Array, target: StreamOrDevice) -> Result<Array> {
        let dims = hidden.shape();
        let dims = dims.as_slice();
        let (b, s, h) = (dims[0], dims[1], dims[2]);
        if s > 1 {
            mlx::ops::indexing::slice_strided(hidden, &[0, s - 1, 0][..], &[b, s, h][..], &[1, 1, 1][..])
        } else {
            Ok(hidden.clone())
        }
    }
}
```

```rust
use base64::Engine;

pub fn decode_image_url(url: &str) -> anyhow::Result<Vec<u8>> {
    if let Some(rest) = url.strip_prefix("data:") {
        // data:image/jpeg;base64,XXX
        let (_meta, b64) = rest
            .split_once(",")
            .ok_or_else(|| anyhow::anyhow!("malformed data URL"))?;
        let bytes = base64::engine::general_purpose::STANDARD.decode(b64)?;
        Ok(bytes)
    } else if url.starts_with("http://") || url.starts_with("https://") {
        // HTTP fetch — blocking, prefill 一次性
        let bytes = reqwest::blocking::get(url)?.bytes()?.to_vec();
        Ok(bytes)
    } else {
        anyhow::bail!("unsupported image_url: {url}")
    }
}
```

//! ```

use std::path::PathBuf;

use mlx::{Array, Dtype};

use ironmlx::core::{generate::{build_position_ids_vl, IMAGE_TOKEN_ID, VISION_START_TOKEN_ID, VISION_END_TOKEN_ID}, Loader};
use ironmlx::models::{Qwen35Model, qwen3_5::image_processor};

const FIXTURE_DIR: &str = "tests/fixtures/p6_qwen35_vl";

fn max_abs_diff(a: &Array, b: &Array) -> f32 {
    let a32 = mlx::ops::cast::astype(a, Dtype::Float32).unwrap();
    let b32 = mlx::ops::cast::astype(b, Dtype::Float32).unwrap();
    let av: Vec<f32> = a32.to_vec().unwrap();
    let bv: Vec<f32> = b32.to_vec().unwrap();
    av.iter().zip(&bv).map(|(a, b)| (a - b).abs()).fold(0.0_f32, f32::max)
}

#[test]
#[ignore]
fn p6_qwen35_vl_logits_match() {
    let dir = std::env::var("QWEN35_MODEL").expect("QWEN35_MODEL");
    let loader = Loader::from_dir(std::path::Path::new(&dir)).unwrap();
    let model = Qwen35Model::from_loader(&loader).unwrap();

    let input_ids: Array = mlx::io::load_npy(&format!("{FIXTURE_DIR}/expected_input_ids.npy")).unwrap();
    let pixel_values_ref: Array = mlx::io::load_npy(&format!("{FIXTURE_DIR}/expected_pixel_values.npy")).unwrap();
    let grid_thw_ref: Array = mlx::io::load_npy(&format!("{FIXTURE_DIR}/expected_image_grid_thw.npy")).unwrap();
    let expected_logits: Array = mlx::io::load_npy(&format!("{FIXTURE_DIR}/expected_last_logits.npy")).unwrap();
    let expected_first_token: i32 = std::fs::read_to_string(&format!("{FIXTURE_DIR}/expected_first_token.txt"))
        .unwrap().trim().parse().unwrap();

    // Convert grid_thw from Array to Vec<(i32,i32,i32)>
    let grids: Vec<i32> = grid_thw_ref.to_vec().unwrap();
    let grids_tup: Vec<(i32, i32, i32)> = grids.chunks_exact(3).map(|c| (c[0], c[1], c[2])).collect();

    // Convert input_ids to i32 slice for position_ids gen
    let input_ids_vec: Vec<i32> = mlx::ops::cast::astype(&input_ids, Dtype::Int32).unwrap().to_vec().unwrap();
    let pos_ids = build_position_ids_vl(&input_ids_vec, &grids_tup, IMAGE_TOKEN_ID, VISION_START_TOKEN_ID, VISION_END_TOKEN_ID, 2);

    // Use HF-prepared pixel_values directly (they should match our preprocess output, verified separately)
    let logits = model.forward_vl(&input_ids, &pos_ids, None, Some(&pixel_values_ref), Some(&grids_tup), ()).unwrap();

    // Strict check 1: logits diff < 0.5
    let diff = max_abs_diff(&logits, &expected_logits);
    assert!(diff < 0.5, "logits max_abs_diff = {diff} (expected < 0.5)");

    // Strict check 2: greedy first token matches
    let dim = logits.shape().as_slice();
    let dim2 = dim.last().copied().unwrap();
    let logits_flat = logits.reshape(&[dim2]).unwrap();
    let our_first = mlx::ops::reduce::argmax(&logits_flat, -1, false).unwrap();
    let our_first: i32 = mlx::ops::cast::astype(&our_first, Dtype::Int32).unwrap().item().unwrap();
    assert_eq!(our_first, expected_first_token, "greedy first token mismatch");

    eprintln!("P6 VL logits match: diff={diff}, first_token={our_first}");
}
