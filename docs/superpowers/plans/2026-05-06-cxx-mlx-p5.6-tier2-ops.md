pub fn nan_to_num_on(
    a: &Array, nan: f32, posinf: Option<f32>, neginf: Option<f32>,
    target: impl Into<crate::StreamOrDevice>,
) -> Result<Array> {
    let (has_pos, pos) = posinf.map_or((false, 0.0), |v| (true, v));
    let (has_neg, neg) = neginf.map_or((false, 0.0), |v| (true, v));
    let (has, dev_only, dt, idx) = target.into().encode();
    let inner = mlx_sys::array::ffi::nan_to_num(
        a.as_inner(), nan, has_pos, pos, has_neg, neg, has, dev_only, dt, idx,
    ).map_err(crate::Error::from)?;
    Ok(Array::from_inner(inner))
}

use crate::{op_with_stream, Array, Result, StreamOrDevice};

op_with_stream! {
    /// Cumulative sum along `axis`. `reverse=true` reverses scan; `inclusive=true` includes element at index.
    pub fn cumsum(a: &Array, axis: i32, reverse: bool, inclusive: bool) -> Result<Array>
        => array::ffi::cumsum(a.as_inner(), axis, reverse, inclusive);

    pub fn cumprod(a: &Array, axis: i32, reverse: bool, inclusive: bool) -> Result<Array>
        => array::ffi::cumprod(a.as_inner(), axis, reverse, inclusive);
